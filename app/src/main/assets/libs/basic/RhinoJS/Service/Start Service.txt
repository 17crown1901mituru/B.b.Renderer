   var Intent = Packages.android.content.Intent;
   // Run through the scripts tab not in the editor
   
    function onCreate(bundle)
   {
      startService();
     
   }

    function startService()
   {
       if (ctx.getAppPath()== null )
      {
         popup("Cant start service in editor");
          return ;
      }
      var intent =  new Intent(ctx,pkg.ScriptService);
      intent.putExtra("path",ctx.getAppPath()+"/Background Service.txt");
      ctx.startService(intent);
   }
