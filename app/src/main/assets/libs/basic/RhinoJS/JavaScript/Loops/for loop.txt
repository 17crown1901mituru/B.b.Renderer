    function onCreate()
   {
      let array = ["One","Two","Three","Four","Five"];
      let len = array.length;
      //Example 1
      for (let i = 0;i<len;i++)
      {
         print(array[i]);
      }
      //Example 2
      for (let i = len-1;i>-1;i--)
      {
         print(array[i]);
      }
      //Example 3
      for (let i = 0;i<len;i++)
      {
          if (i == 2) continue ;
         print(array[i]);
      }
      //Example 4
      for (let i = 0;i<len;i++)
      {
          if (i == 3) break ;
         print(array[i]);
      }
      
   }
