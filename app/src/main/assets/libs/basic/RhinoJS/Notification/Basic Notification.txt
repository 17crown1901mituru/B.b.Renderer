   var Context = Packages.android.content.Context;
   var Notification = Packages.android.app.Notification;
   var NotificationManager = Packages.android.app.NotificationManager;
   var Intent = Packages.android.content.Intent;
   var PendingIntent = Packages.android.app.PendingIntent;
   var BitmapFactory = Packages.android.graphics.BitmapFactory;
   var NotificationChannel = Packages.android.app.NotificationChannel;
   var notificationChannel =  null ;
   var notificationManager = ctx.getSystemService(Context.NOTIFICATION_SERVICE);
   var notificationId = "1234";
   
   notify("Notification title","Notification message",notificationId);
   
    function notify(title,text,id)
   {
       if (android.os.Build.VERSION.SDK_INT>=26)
      {
          if (notificationChannel== null )
         {
            notificationChannel =  new NotificationChannel(id,"TestNotify",NotificationManager.IMPORTANCE_DEFAULT);
            notificationChannel.setDescription("Test notifications");
            notificationManager.createNotificationChannel(notificationChannel);
         }
         
      }
      var intent =  new Intent(ctx,pkg.Plugins);
      intent.putExtra("scriptPath",ctx.getScriptPath());
      intent.setFlags(Intent.FLAG_ACTIVITY_BROUGHT_TO_FRONT);
      var pintent = PendingIntent.getActivity(ctx,0,intent,PendingIntent.FLAG_UPDATE_CURRENT);
      var builder =  new Notification.Builder(ctx)
      .setContentTitle(title)
      .setContentText(text)
      .setAutoCancel( true )
      .setTicker(title)
      .setSmallIcon(pkg.R.drawable.edit)
     // .setLargeIcon(BitmapFactory.decodeResource(ctx.getResources(),pkg.R.drawable.ic_launcher))//.setColor(getResources().getColor(R.color.actionBar))
      .setContentIntent(pintent);
       if (notificationChannel!= null )builder.setChannelId(id);
      var notification = builder.build();
      notificationManager.notify(id,notification);
   }
   
   function onDestroy(){
      notificationManager.cancel(notificationId);
   }
