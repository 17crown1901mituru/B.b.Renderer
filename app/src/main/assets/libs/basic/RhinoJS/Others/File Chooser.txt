   //selectFile(true)
   //selectFile(false,true)
    function selectFile(selectDir,multiple)
   {
      var fc =  new pkg.utils.FileChooser();
      fc.setTitle("Select file");
      fc.addFilter(".txt");
      fc.addFilter(".zip");
      fc.addFilter(".jpg");
      fc.addFilter(".mp3");
       if (selectDir)fc.setSelectFolder( true );
      fc.setRootPath(pkg.Globals.externalPath);
       if (!multiple)fc.explore(ctx,pkg.Globals.externalPath);
       else fc.multiExplore(ctx,pkg.Globals.externalPath);
      fc.setOnSelectionListener( function (mpath,isDir)
      {
         popup("Path: " + mpath + "\nIsDirectory: " + isDir);
      });
      fc.setOnMultiSelectionListener( function (paths)
      {
         popup("Paths: " + jsArray(paths));
      });
   }
