   var Intent = Packages.android.content.Intent;
   var IntentFilter = Packages.android.content.IntentFilter;
   var BatteryManager = Packages.android.os.BatteryManager;
   var TextView = Packages.android.widget.TextView;
   let textview,br;

    function onCreate(bundle)
   {
      ctx.setTitle("Battery info");
      textview =  new TextView(ctx);
      textview.setPadding(20,20,0,0);
      br =  new pkg.utils.JASBroadcastReceiver();
      br.setListener( function (context,intent)
      {
         let str = "";
         str += "Level: " + intent.getIntExtra(BatteryManager.EXTRA_LEVEL,0)+"%\n";
         str += "Voltage: " + intent.getIntExtra(BatteryManager.EXTRA_VOLTAGE,0)+"mV\n";
         str += "Temperature: " + intent.getIntExtra(BatteryManager.EXTRA_TEMPERATURE,0)/10 + "°C\n";
         str += "Technology: " + intent.getExtras().getString(BatteryManager.EXTRA_TECHNOLOGY)+"\n";
         textview.setText(str);
      });
      ctx.setContentView(textview);
      ctx.registerReceiver(br, new IntentFilter(Intent.ACTION_BATTERY_CHANGED));
   }

    function onDestroy()
   {
      ctx.unregisterReceiver(br);
   }
