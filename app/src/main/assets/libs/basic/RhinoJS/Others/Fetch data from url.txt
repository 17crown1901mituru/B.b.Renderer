   var InputStreamReader = Packages.java.io.InputStreamReader;
   var BufferedReader = Packages.java.io.BufferedReader;
   var StringBuilder = Packages.java.lang.StringBuilder;
   var URL = Packages.java.net.URL;

   
   
   getUrlFileContent("https://google.com", function (result)
   {
      if(result)alert(result);
   });

    function getUrlFileContent(url,callback)
   {
      var result;
      pkg.Utils.showProgress(ctx,"Loading");
      bTask( function ()
      {
         result = remoteText(url);
      }
      , function ()
      {
         pkg.Utils.hideProgress();
         callback(result);
      });

       function remoteText(url)
      {
         try 
         {
            var request_url =  new URL(url);
            var http_conn = request_url.openConnection();
            http_conn.setFollowRedirects( true );
            http_conn.setConnectTimeout(10000);
            http_conn.setReadTimeout(10000);
            http_conn.setInstanceFollowRedirects( true );
            var br =  new BufferedReader( new InputStreamReader(http_conn.getInputStream()));
            var sb =  new StringBuilder();
            var output;
            while ((output = br.readLine())!= null )
            {
               sb.append(output+"\n");
            }
            br.close();
             return sb.toString();
         }
          catch (e)
         {
            popup(e);
             return "";
         }
         
      }
      
   }
