   var View = Packages.android.view.View;
   var Bitmap = Packages.android.graphics.Bitmap;
   var ThumbnailUtils = Packages.android.media.ThumbnailUtils;
   var Settings = Packages.android.provider.Settings;
   var Uri = Packages.android.net.Uri;
   var Color = Packages.android.graphics.Color;
   var LinearLayout = Packages.android.widget.LinearLayout;
   var SearchView = Packages.android.widget.SearchView;
   var BitmapFactory = Packages.android.graphics.BitmapFactory;
   var KeyEvent = Packages.android.view.KeyEvent;
   var CustomGridAdapter = pkg.adapters.CustomGridAdapter;
   var LayoutParams = Packages.android.widget.LinearLayout.LayoutParams;
   var Intent = Packages.android.content.Intent;
   var ApplicationInfo = Packages.android.content.pm.ApplicationInfo;
   var R = pkg.R;
   var mpkg = [];
   var label = [];
   var pkgnm = [];
   var tmpkg = [];
   var tlabel = [];
   var tpkgnm = [];
   var msc = [];
   var mscu = [];
   var msca = [];
   var selectedPositions = [];
   var selectApps =  false ;
   var lay,edt,gridView,query,cd;
   var intent;
   var title = "Installed Apps";
   var width = pkg.Utils.getDisplayWidth(ctx);

    function onCreate(savedInstanceState)
   {
      ctx.setTitle(title);
      ctx.setTheme(android.R.style.Theme_DeviceDefault_Wallpaper);
      setTaskDescription(title,BitmapFactory.decodeFile(ctx.getAppPath()+"/icon.png"),Color.parseColor("#32174b"));
      setStatusBarBackgroundColor(Color.parseColor("#1e32174b"));
      pkg.Utils.setNavigationBarBackgroundColor(ctx,Color.BLACK);
      ctx.getActionBar().setBackgroundDrawable( new android.graphics.drawable.ColorDrawable(Color.parseColor("#1e32174b")));
      cd =  new CustomGridAdapter(ctx,label,mpkg);
      lay =  new LinearLayout(ctx);
      lay.setLayoutParams( new LayoutParams(LayoutParams.FILL_PARENT,LayoutParams.FILL_PARENT,1));
      lay.setOrientation(1);
      lay.setGravity(android.view.Gravity.TOP);
      query = 
      {
         onQueryTextSubmit: function (s)
         {
             return  false ;
         }
         ,onQueryTextChange: function (s)
         {
            msc = [];
            mscu = [];
            msca = [];
            for (var i = 0;i<label.length;i++)
            {
                if (label[i].toLowerCase().search(s.toLowerCase())!=-1)
               {
                  msc.push(label[i]);
                  mscu.push(mpkg[i]);
                  msca.push(pkgnm[i]);
               }
               
            }
            selectedPositions = [];
            cd.setList(msc,mscu);
             return  false ;
         }
         
      };
      edt =  new SearchView(ctx);
      edt.setOnQueryTextListener(query);
      edt.setLayoutParams( new LayoutParams(LayoutParams.WRAP_CONTENT,LayoutParams.WRAP_CONTENT,1));
      edt.setBackgroundColor(Color.TRANSPARENT);
      ctx.getActionBar().setCustomView(edt);
      ctx.getActionBar().setDisplayShowCustomEnabled( true );
      gridView =  new android.widget.GridView(ctx);
      gridView.setNumColumns(4);
      cd.setImageType("drawable");
      cd.setContentWidth(width/5);
      cd.setImageHeight(width/5.5);
      cd.setBackgroundColor(android.graphics.Color.parseColor("#1e141414"));
      cd.setTextColor(android.graphics.Color.WHITE);
      cd.setTextSize(14);
      onConfigurationChanged();
      gridView.setAdapter(cd);
      gridView.setOnItemClickListener( function (parent,v,position,id)
      {
         launch(position);
      });
      gridView.setOnItemLongClickListener( function (parent,v,position,id)
      {
         appSettings(position);
          return  true ;
      });
      bTask( function ()
      {
         apps();
      }
      , function ()
      {
         cd.setList(label,mpkg);
      });
      lay.addView(gridView);
      ctx.setContentView(lay);
      //blurImage(ctx.getWallpaper().getBitmap());
      
   }

    function appSettings(pos)
   {
      var i =  new Intent();
      i.setAction(Settings.ACTION_APPLICATION_DETAILS_SETTINGS);
      i.setData(Uri.fromParts("package",pkgnm[pos], null ));
      ctx.startActivity(i);
   }

    function launch(pos)
   {
       if (msca.length>0)
      {
         intent = ctx.getPackageManager().getLaunchIntentForPackage(msca[pos]);
      }
       else 
      {
         intent = ctx.getPackageManager().getLaunchIntentForPackage(pkgnm[pos]);
      }
       if (intent !=  null )
      {
         ctx.startActivity(intent);
         ctx.overridePendingTransition(android.R.anim.slide_in_left,android.R.anim.slide_out_right);
      }
       else 
      {
         popup("Unable to launch ");
      }
      
   }

    function apps()
   {
      var androidPackages = ctx.getPackageManager().getInstalledPackages(1);
      var pm = ctx.getPackageManager();
      var intent =  new Intent(Intent.ACTION_MAIN, null );
      intent.addCategory(Intent.CATEGORY_LAUNCHER);
      var tpkgs = pm.queryIntentActivities(intent,1);
      runOnUIThread( function ()
      {
         pkg.Utils.popup(ctx,"Loading apps...",Color.WHITE,Color.BLACK,5000);
      });
      for (var i = 0;i<tpkgs.size();i++)
      {
         tmpkg.push(tpkgs.get(i).activityInfo.loadIcon(ctx.getPackageManager()));
         tlabel.push(tpkgs.get(i).activityInfo.loadLabel(ctx.getPackageManager()).toString());
         tpkgnm.push(tpkgs.get(i).activityInfo.packageName);
         runOnUIThread( function ()
         {
            ctx.setTitle(title+" ("+((i/tpkgs.size())*100).toFixed(0)+"%)");
         });
          if (i%20 == 0)
         {
            mpkg = tmpkg;
            label = tlabel;
            pkgnm = tpkgnm;
            runOnUIThread( function ()
            {
               cd.setList(label,mpkg);
            });
         }
         
      }
      sortFinalize();
      runOnUIThread( function ()
      {
         pkg.Utils.popup(ctx,"Completed",Color.WHITE,Color.BLACK,4000);
         ctx.setTitle(title);
      });
   }

    function sortFinalize()
   {
      var multi = [];
      for (i = 0;i<tlabel.length;i++)
      {
         multi[i] = [tlabel[i],tmpkg[i],tpkgnm[i]];
      }
      multi = multi.sort();
      tlabel =  null ;
      tmpkg =  null ;
      tpkgnm =  null ;
      for (i = 0;i<multi.length;i++)
      {
         label[i] = multi[i][0];
         mpkg[i] = multi[i][1];
         pkgnm[i] = multi[i][2];
      }
      
   }

    function onResume()
   {
      ctx.overridePendingTransition(android.R.anim.slide_in_left,android.R.anim.slide_out_right);
      try 
      {
         ctx.getWindow().getDecorView().setSystemUiVisibility(View.SYSTEM_UI_FLAG_HIDE_NAVIGATION|View.SYSTEM_UI_FLAG_IMMERSIVE);
      }
       catch (e)
      {
         
      }
      
   }

    function onKeyUp(keycode,event)
   {
       if (!event.isCanceled())
      {
          if (keycode === KeyEvent.KEYCODE_BACK)
         {
             return  false ;
         }
         
      }
       return  false ;
   }

    function blurImage(bitmap)
   {
      var bmp = ThumbnailUtils.extractThumbnail(bitmap,2,2);
      var tlay = lay.getParent().getParent().getParent();
      tlay.setBackgroundDrawable(android.graphics.drawable.BitmapDrawable(ctx.getResources(),bmp));
      var pixelColor = bmp.getPixel(0,0);
      setStatusBarBackgroundColor(pixelColor);
      setTaskDescription("JASLauncher", null ,pixelColor);
   }

    function onConfigurationChanged()
   {
      width = ctx.getWindowManager().getDefaultDisplay().getWidth();
       if (width<300)columns = 3;
       else  if (width<700)columns = 4;
       else  if (width<1100)columns = 5;
       else  if (width<1500)columns = 6;
       else  if (width<1900)columns = 7;
       else  if (width<2300)columns = 8;
       else  if (width<2700)columns = 9;
       else  if (width<3100)columns = 10;
       else  if (width<3500)columns = 11;
       else columns = Math.floor(width/160);
      cd.setContentWidth((width/columns)-(width/45));
      cd.setImageHeight((width/columns)-(width/15));
      gridView.setNumColumns(columns);
      cd.notifyDataSetChanged();
   }
