   var AbsListView = Packages.android.widget.AbsListView;
   var LinearLayout = Packages.android.widget.LinearLayout;
   var ArrayAdapter = Packages.android.widget.ArrayAdapter;
   var ListView = Packages.android.widget.ListView;
   var TextView = Packages.android.widget.TextView;
   var Gravity = Packages.android.view.Gravity;
   var Color = Packages.android.graphics.Color;
   var LayoutParams = Packages.android.widget.LinearLayout.LayoutParams;
   var PaintDrawable = Packages.android.graphics.drawable.PaintDrawable;
   var titles = ["1 Title","2 Title","3 Title","4 Title"];
   var details = ["Detail 1","Detail 2","Detail 3","Detail 4"];
   var color = Color.parseColor("#2368a6");
   ctx.setTheme(android.R.style.Theme_DeviceDefault_Light_DarkActionBar);

    function onCreate(bundle)
   {
      pkg.Utils.setActionBarBackgroundColor(ctx,color);
      pkg.Utils.setStatusBarBackgroundColor(ctx,color);
      pkg.Utils.setNavigationBarBackgroundColor(ctx,color);
      ctx.setTitle("Custom ListView");
      lay =  new LinearLayout(ctx);
      lay.setPadding(10,10,10,10);
      lay.setOrientation(LinearLayout.VERTICAL);
      adapter =  new pkg.adapters.CustomListAdapter();
      adapter.setOnGetView( function (position,convertView,parent)
      {
          if (convertView ==  null )
         {
            convertView = getCustomLayout();
         }
         var tv = convertView.findViewWithTag("txt");
         var tv1 = convertView.findViewWithTag("txt1");
         var tv2 = convertView.findViewWithTag("txt2");
         var tv3 = convertView.findViewWithTag("txt3");
         tv1.setText(titles[position]);
         tv3.setText(details[position]);
         tv.setText(titles[position].slice(0,1));
          return convertView;
      });
      listview =  new ListView(ctx);
      listview.setAdapter(adapter);
      listview.setOnItemClickListener( function (parent,view,position,id)
      {
         popup(parent.getItemAtPosition(position));
      });
      lay.addView(listview);
      adapter.setList(titles,details);
      ctx.setContentView(lay);
      setTimeout( function ()
      {
         changeValues();
      }
      ,3000);
   }

    function changeValues()
   {
      titles[2] = "Changed Title";
      details[3] = "Changed Detail";
      adapter.notifyDataSetChanged();
   }

    function getCustomLayout()
   {
      var lay =  new LinearLayout(ctx);
      lay.setLayoutParams( new AbsListView.LayoutParams(LayoutParams.MATCH_PARENT,LayoutParams.MATCH_PARENT,10));
      lay.setPadding(10,10,10,10);
      var txt =  new TextView(ctx);
      txt.setTag("txt");
      txt.setGravity(Gravity.CENTER);
      txt.setTextSize(20);
      txt.setPadding(5,10,5,10);
      txt.setTextColor(Color.WHITE);
      txt.setBackgroundDrawable(itemBackground());
      txt.setLayoutParams( new LayoutParams(LayoutParams.MATCH_PARENT,LayoutParams.MATCH_PARENT,8));
      txt.getLayoutParams().setMargins(0,0,10,0);
      lay.addView(txt);
      var lay1 =  new LinearLayout(ctx);
      lay1.setLayoutParams( new LayoutParams(LayoutParams.MATCH_PARENT,LayoutParams.MATCH_PARENT,2));
      lay1.setOrientation(1);
      lay.addView(lay1);
      var lay2 =  new LinearLayout(ctx);
      lay2.setLayoutParams( new LayoutParams(LayoutParams.MATCH_PARENT,LayoutParams.WRAP_CONTENT,10));
      lay2.setOrientation(0);
      lay1.addView(lay2);
      var txt1 =  new TextView(ctx);
      txt1.setTag("txt1");
      txt1.setTextSize(16);
      txt1.setMaxLines(1);
      txt1.setPadding(5,5,5,5);
      txt1.setGravity(Gravity.CENTER_VERTICAL);
      txt1.setLayoutParams( new LayoutParams(LayoutParams.MATCH_PARENT,LayoutParams.MATCH_PARENT,3));
      lay2.addView(txt1);
      var txt2 =  new TextView(ctx);
      txt2.setTag("txt2");
      txt2.setMaxLines(1);
      txt2.setTextSize(20);
      txt2.setGravity(Gravity.CENTER_VERTICAL);
      txt2.setPadding(5,5,5,5);
      txt2.setLayoutParams( new LayoutParams(LayoutParams.MATCH_PARENT,LayoutParams.MATCH_PARENT,7));
      var txt3 =  new TextView(ctx);
      txt3.setTag("txt3");
      txt3.setMaxLines(3);
      txt3.setTextSize(18);
      txt3.setGravity(Gravity.CENTER_VERTICAL);
      txt3.setPadding(5,5,5,5);
      txt3.setLayoutParams( new LayoutParams(LayoutParams.MATCH_PARENT,LayoutParams.WRAP_CONTENT,0));
      lay1.addView(txt3);
       return lay;
   }

    function itemBackground()
   {
      var pd =  new PaintDrawable(color);
      pd.setCornerRadii([0,0,50,50,0,0,50,50]);
       return pd;
   }
