   var RelativeLayout = Packages.android.widget.RelativeLayout;
   var Button = Packages.android.widget.Button;
   var Color = Packages.android.graphics.Color;
   var TextView = Packages.android.widget.TextView;
   var Gravity = Packages.android.view.Gravity;
   var color = Color.parseColor("#236896");
   ctx.setTheme(android.R.style.Theme_DeviceDefault_Light_DarkActionBar);
    var lay,btn;
    function onCreate()
   {
      pkg.Utils.setActionBarBackgroundColor(ctx,color);
      pkg.Utils.setStatusBarBackgroundColor(ctx,color);
      pkg.Utils.setNavigationBarBackgroundColor(ctx,color);
      ctx.setTitle("Button");
      lay =  new RelativeLayout(ctx);
      lay.setGravity(Gravity.CENTER);
      btn =  new Button(ctx);
      btn.setLayoutParams( new RelativeLayout.LayoutParams(RelativeLayout.LayoutParams.WRAP_CONTENT,RelativeLayout.LayoutParams.WRAP_CONTENT));
      btn.setText("Button");
      btn.setOnClickListener( function ()
      {
         popup("Clicked");
      });
      btn.setOnLongClickListener( function ()
      {
         popup("Long clicked");
          return  true ;
      });
      btn.setPadding(30,30,30,30);
      btn.setGravity(Gravity.CENTER);
      lay.addView(btn);
      ctx.setContentView(lay);
   }
