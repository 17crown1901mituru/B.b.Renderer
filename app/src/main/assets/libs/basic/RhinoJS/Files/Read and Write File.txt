    function readFile(file)
   {
      var br =  new BufferedReader( new FileReader(file));
      var sb =  new StringBuilder();
      try 
      {
         var line = br.readLine();
         while (line !=  null )
         {
            sb.append(line);
            sb.append("\n");
            line = br.readLine();
         }
         
      }
       catch (e)
      {
         e.printStackTrace();
      }
       finally 
      {
         br.close();
      }
       return sb.toString();
   }

    function writeFile(file,content)
   {
      var bw =  new BufferedWriter( new FileWriter(file));
      try 
      {
         bw.write(content);
         bw.flush();
      }
       catch (e)
      {
         e.printStackTrace();
      }
       finally 
      {
         bw.close();
      }
      
   }
