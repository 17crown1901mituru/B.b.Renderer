   var ViewPropertyAnimator = Packages.android.view.ViewPropertyAnimator;
   var ValueAnimator = Packages.android.animation.ValueAnimator;
   var Color = Packages.android.graphics.Color;
   var Gravity = Packages.android.view.Gravity;
   var ImageView = Packages.android.widget.ImageView;
   var LinearLayout = Packages.android.widget.LinearLayout;
   var color = Color.parseColor("#236846");

    function onCreate(bundle)
   {
      pkg.Utils.setActionBarBackgroundColor(ctx,color);
      pkg.Utils.setStatusBarBackgroundColor(ctx,color);
      pkg.Utils.setNavigationBarBackgroundColor(ctx,color);
      ctx.setTitle("Scale Animation");
      lay =  new LinearLayout(ctx);
      lay.setBackgroundColor(Color.WHITE);
      lay.setGravity(Gravity.CENTER);
      image =  new ImageView(ctx);
      image.setScaleX(-1);
      image.setScaleY(-1);
      image.setLayoutParams( new LinearLayout.LayoutParams(LinearLayout.LayoutParams.WRAP_CONTENT,LinearLayout.LayoutParams.WRAP_CONTENT,0));
      image.setImageResource(pkg.R.drawable.ic_launcher);
      lay.addView(image);
      ctx.setContentView(lay);
      //ViewPropertyAnimator anim;
      var anim = image.animate();
      anim.setListener( new AnimationListener());
      anim.scaleX(1).scaleY(1).setDuration(3000).start();
   }
