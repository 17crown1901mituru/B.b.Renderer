   var DialogInterface = Packages.android.content.DialogInterface;
   var Builder = Packages.android.app.AlertDialog.Builder;
   showDialog("Alert dialog message","Alert");

    function showDialog(message,title)
   {
      var builder =  new Builder(ctx);
      builder.setMessage(message).setTitle(title).setPositiveButton("Ok", function (di)
      {
         //DialogInterface di;
         popup("Ok clicked");
         di.dismiss();
      }).setNegativeButton("Cancel", function (di)
      {
         popup("Cancel clicked");
         di.dismiss();
      }).show();
   }
