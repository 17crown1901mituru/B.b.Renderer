    function onCreate(a,b,c)
   {
      print("onCreate");
      showToast("Service created");
   }

    function onStartCommand(a,b,c)
   {
      print("onStartCommand");
      showToast("onStart Command");
   }

    function onDestroy()
   {
      print("onDestroy");
      showToast("Service destroyed");
   }

    function showToast(message)
   {
      Toast.makeText(ctx.getApplicationContext(),message,Toast.LENGTH_LONG).show();
   }
