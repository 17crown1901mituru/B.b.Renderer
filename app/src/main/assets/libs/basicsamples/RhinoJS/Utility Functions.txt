   // prompt - shows a dialog with a box to input text
   var name = prompt("Enter name","");
   popup(name);
   var age = prompt("Enter your age","Default text","Custom title");
   popup(age);
   // confirm - shows a confirmation dialog
   var canExit = confirm("Do you want to exit?");
   popup(canExit);
   var canDelete = confirm("Do you want to delete this file?","Delete File");
   popup(canDelete);
   // select - shows a dialog with list of items to select one.
   var selected = select(["One","Two","Three","Four","Five"]);
   popup(selected);
   // choose - same as select
   var choosen = choose(["One","Two","Three","Four","Five"],2,1,"Single Choice");
   popup(choosen);
   // multichoice - choose more than one item
   var choosenArray = multichoice(["One","Two","Three","Four","Five"],[1,3],1,"Multi Choice");
   choosenArray = jsArray(choosenArray);
   popup(choosenArray);
   // range - choose a number from a range
   var choosenNum = range("Change volume",50,20,28,"Volume");
   popup(choosenNum);
