

   var Application = Packages.android.app.Application;
   var LayoutParams = Packages.android.widget.LinearLayout.LayoutParams;
   var TextView = Packages.android.widget.TextView;
   var WebView = Packages.android.webkit.WebView;
   var LinearLayout = Packages.android.widget.LinearLayout;
   var width = pkg.Utils.getDisplayWidth(ctx);
    if (pkg.Utils.getApiLevel()>=28)
   {
       if (ctx.getPackageName() !== Application.getProcessName())
      {
         WebView.setDataDirectorySuffix("simplewebview");
      }
      
   }
   ctx.setTheme(android.R.style.Theme_DeviceDefault_Light);

    function onCreate(bundle)
   {
      ctx.setTitle("WebView");
      lay =  new LinearLayout(ctx);
      lay.setLayoutParams( new LayoutParams(LayoutParams.FILL_PARENT,LayoutParams.FILL_PARENT,0));
      lay.setOrientation(LinearLayout.VERTICAL);
      prog =  new TextView(ctx);
      prog.setBackgroundResource(pkg.R.color.actionBar);
      prog.setLayoutParams( new LayoutParams(0,5));
      wb =  new WebView(ctx);
      wb.setLayoutParams( new LayoutParams(LayoutParams.FILL_PARENT,LayoutParams.FILL_PARENT,0));
      wb.setVerticalScrollBarEnabled( true );
      wb.loadUrl("https://www.google.com");
      var set = wb.getSettings();
      set.setAllowFileAccessFromFileURLs( true );
      set.setUseWideViewPort( true );
      set.setLoadWithOverviewMode( true);
      set.setJavaScriptEnabled( true );
      set.setLoadsImagesAutomatically( true );
      set.setJavaScriptCanOpenWindowsAutomatically( false );
      set.setSupportMultipleWindows( false );
      set.setAppCacheEnabled( false );
      set.setBuiltInZoomControls( true );
      set.setDisplayZoomControls( false );
      
      
      //Avoid using below classes, they may be removed in future. The recommended way is extending WebViewClient and WebChomeClient classes with JavaAdapter
      var wcc =  new pkg.web.JASWebChromeClient();
      var wvc =  new pkg.web.JASWebViewClient();
      wcc.onWebChromeActions( function (method,params)
      {
         switch (method)
         {
            case "onReceivedTitle":ctx.setTitle(params[1]);
             break ;
            case "onProgressChanged":prog.setLayoutParams(new LayoutParams((params[1]/100)*width,5));
             break ;
             default :
         }
         
      });
      wvc.onWebViewActions( function (method,params)
      {
         switch (method)
         {
            case "onPageStarted": break ;
            case "onPageFinished": break ;
             default :
         }
         
      });
      wvc.setShouldOverrideUrlLoading( function (name,params)
      {
         
      });
      wb.setWebViewClient(wvc);
      wb.setWebChromeClient(wcc);
      lay.addView(prog);
      lay.addView(wb);
      ctx.setContentView(lay);
   }
