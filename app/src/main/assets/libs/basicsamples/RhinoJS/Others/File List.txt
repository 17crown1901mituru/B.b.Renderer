   var Environment = Packages.android.os.Environment;
   
   var sdDir = Environment.getExternalStorageDirectory();
   //sdDir = ctx.getExternalFilesDir(null); //Uncomment this line in android 10+ device
   var sdFiles = sdDir.listFiles();
   var len = sdFiles.length;
   var filesStr = "";
   for (var i = 0;i<len;i++)
   {
      filesStr += sdFiles[i].getName()+"\n";
   }
   alert(filesStr);
   ctx.finish();
