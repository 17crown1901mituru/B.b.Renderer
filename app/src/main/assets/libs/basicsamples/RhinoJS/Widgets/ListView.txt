   var Color = Packages.android.graphics.Color;
   var LinearLayout = Packages.android.widget.LinearLayout;
   var Arrays = Packages.java.util.Arrays;
   var ArrayAdapter = Packages.android.widget.ArrayAdapter;
   var ListView = Packages.android.widget.ListView;
   let lay,listview,adapter,listview2,adapter2;
   
      var color = Color.parseColor("#236896");
   ctx.setTheme(android.R.style.Theme_DeviceDefault_Light_DarkActionBar);
   
   let items = ["Item1","Item2","Item3","Item4"];
   let subitems = ["SubItem1","SubItem2","SubItem3","SubItem4"];
    function onCreate(bundle)
   {
      setActionBarBackgroundColor(color);
      setStatusBarBackgroundColor(color);
      pkg.Utils.setNavigationBarBackgroundColor(ctx,color);
      
      ctx.setTitle("ListView");
      lay =  new LinearLayout(ctx);
      lay.setPadding(20,20,0,0);
      lay.setOrientation(LinearLayout.VERTICAL);
      // Default list adapter
      adapter =  new ArrayAdapter(ctx,pkg.R.layout.list_item_text);
      listview =  new ListView(ctx);
      listview.setAdapter(adapter);
      listview.setOnItemClickListener(function(parent,view,position,id){
         popup(parent.getItemAtPosition(position));
      });
      lay.addView(listview);
      adapter.addAll(Arrays.asList(items));
      //CustomAdapter
      adapter2 =  new pkg.adapters.CustomListAdapter(ctx,items,subitems);
      listview2 =  new ListView(ctx);
      listview2.setAdapter(adapter2);
      lay.addView(listview2);
      
      //Custom view -- Most recommended
      //See Custom Listview Sample
      
      ctx.setContentView(lay);
   }
