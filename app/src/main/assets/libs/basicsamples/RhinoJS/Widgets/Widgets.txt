   var RadioGroup = Packages.android.widget.RadioGroup;
   var RadioButton = Packages.android.widget.RadioButton;
   var ScrollView = Packages.android.widget.ScrollView;
   var Switch = Packages.android.widget.Switch;
   var RatingBar = Packages.android.widget.RatingBar;
   var CalendarView = Packages.android.widget.CalendarView;
   var LayoutParams = Packages.android.widget.LinearLayout.LayoutParams;
   var CheckBox = Packages.android.widget.CheckBox;
   var ToggleButton = Packages.android.widget.ToggleButton;
   var SeekBar = Packages.android.widget.SeekBar;
   var EditText = Packages.android.widget.EditText;
   var Button = Packages.android.widget.Button;
   var Color = Packages.android.graphics.Color;
   var TextView = Packages.android.widget.TextView;
   var Gravity = Packages.android.view.Gravity;
   var LinearLayout = Packages.android.widget.LinearLayout;
   var color = Color.parseColor("#236896");
   ctx.setTheme(android.R.style.Theme_DeviceDefault_Light_DarkActionBar);

    function onCreate()
   {
      pkg.Utils.setActionBarBackgroundColor(ctx,color);
      pkg.Utils.setStatusBarBackgroundColor(ctx,color);
      pkg.Utils.setNavigationBarBackgroundColor(ctx,color);
      ctx.setTitle("Widgets");
      sc = new ScrollView(ctx);
      sc.setFillViewport(true);
      lay =  new LinearLayout(ctx);
      lay.setPadding(30,30,30,30);
      lay.setLayoutParams( new LayoutParams(LayoutParams.FILL_PARENT,LayoutParams.FILL_PARENT,0));
      lay.setOrientation(LinearLayout.VERTICAL);
      btn =  new Button(ctx);
      btn.setLayoutParams( new LayoutParams(LayoutParams.WRAP_CONTENT,LayoutParams.WRAP_CONTENT,0));
      btn.setOnClickListener( function ()
      {
         popup("Button clicked");
      });
      btn.setText("Button");
      lay.addView(btn);
      tv =  new TextView(ctx);
      tv.setLayoutParams( new LayoutParams(LayoutParams.WRAP_CONTENT,LayoutParams.WRAP_CONTENT,0));
      tv.setText("TextView");
      lay.addView(tv);
      sw = new Switch(ctx);
      sw.setChecked(true);
      lay.addView(sw);
      chk =  new CheckBox(ctx);
      chk.setLayoutParams( new LayoutParams(LayoutParams.WRAP_CONTENT,LayoutParams.WRAP_CONTENT,0));
      chk.setText("CheckBox");
      chk.setChecked( true );
      lay.addView(chk);
      tgl =  new ToggleButton(ctx);
      tgl.setLayoutParams( new LayoutParams(LayoutParams.WRAP_CONTENT,LayoutParams.WRAP_CONTENT,0));
      tgl.setText("Toggle Button");
      lay.addView(tgl);
      skb =  new SeekBar(ctx);
      skb.setLayoutParams( new LayoutParams(LayoutParams.FILL_PARENT,LayoutParams.WRAP_CONTENT,0));
      skb.setMax(100);
      skb.setProgress(50);
      lay.addView(skb);
      edt =  new EditText(ctx);
      edt.setLayoutParams( new LayoutParams(LayoutParams.FILL_PARENT,LayoutParams.WRAP_CONTENT,0));
      edt.setText("Edit Text");
      lay.addView(edt);
      rd = new RadioButton(ctx);
      rd.setChecked(false);
      rg = new RadioGroup(ctx)
      rg.addView(rd);
      lay.addView(rg);
      rt = new RatingBar(ctx);
      rt.setNumStars(8);
      rt.setRating(5);
      lay.addView(rt);
      cl =  new CalendarView(ctx);
      lay.addView(cl);
      sc.addView(lay);
      ctx.setContentView(sc);
   }
