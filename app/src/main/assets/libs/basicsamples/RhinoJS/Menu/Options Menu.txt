   var Color = Packages.android.graphics.Color;
var Menu = Packages.android.view.Menu;

   ctx.setTheme(android.R.style.Theme_DeviceDefault_Light);
   setStatusBarBackgroundColor(Color.parseColor("#236846"));
    function onCreate()
   {
      ctx.setTitle("Options Menu");
   }
    function onPrepareOptionsMenu(menu)
   {
      menu.clear();
      menu.add(Menu.NONE,1,Menu.NONE,"Option 1");
      menu.add(Menu.NONE,2,Menu.NONE,"Option 2");
      menu.add(Menu.NONE,3,Menu.NONE,"Exit");
   }
    function onOptionsItemSelected(item)
   {
      switch (item.getItemId())
      {
         case 1:popup("Clicked Option 1");
          break ;
         case 2:popup("Clicked Option 2");
          break ;
         case 3:ctx.finish();
          break ;
          default :
      }
      
   }
