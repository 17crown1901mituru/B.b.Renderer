   var InputType = Packages.android.text.InputType;
   var MediaPlayer = Packages.android.media.MediaPlayer;
   var MediaController = Packages.android.widget.MediaController;
   var Intent = Packages.android.content.Intent;
   var VideoView = Packages.android.widget.VideoView;
   var Gravity = Packages.android.view.Gravity;
   var Button = Packages.android.widget.Button;
   var LinearLayout = Packages.android.widget.LinearLayout;
   var Color = Packages.android.graphics.Color;
   var color = Color.parseColor("#7685a7");
   var rq = 63754;
   var lay,btn,med;
   ctx.setTheme(android.R.style.Theme_DeviceDefault);

    function onCreate()
   {
      ctx.setTitle("Audio Player");
      pkg.Utils.setActionBarBackgroundColor(ctx,color);
      pkg.Utils.setStatusBarBackgroundColor(ctx,color);
      lay =  new LinearLayout(ctx);
      lay.setBackgroundColor(Color.BLACK);
      lay.setOrientation(LinearLayout.VERTICAL);
      lay.setGravity(Gravity.CENTER_HORIZONTAL);
      btn =  new Button(ctx);
      var params =  new LinearLayout.LayoutParams(LinearLayout.LayoutParams.WRAP_CONTENT,LinearLayout.LayoutParams.WRAP_CONTENT,0);
      params.setMargins(20,30,20,30);
      btn.setLayoutParams(params);
      btn.setInputType(InputType.TYPE_TEXT_FLAG_CAP_WORDS);
      btn.setOnClickListener( function ()
      {
         pickFile();
      });
      btn.setText("Pick an audio");
      lay2 =  new LinearLayout(ctx);
      lay2.setGravity(Gravity.CENTER);
      lay2.setLayoutParams( new LinearLayout.LayoutParams(LinearLayout.LayoutParams.MATCH_PARENT,LinearLayout.LayoutParams.MATCH_PARENT,0));
      lay.addView(btn);
      ctx.setContentView(lay);
   }

    function pickFile()
   {
      var intent =  new Intent(Intent.ACTION_GET_CONTENT);
      intent.setType("audio/*");
      ctx.startActivityForResult(intent,rq);
   }

    function playAudio(src)
   {
      try 
      {
          if (med)
         {
            med.stop();
            med.release();
         }
         med =  new MediaPlayer();
         med.setDataSource(ctx,src);
         med.prepare();
         med.start();
         med.setOnCompletionListener( function ()
         {
            playAudio(src);
         });
      }
       catch (e)
      {
         popup(e);
      }
      
   }

    function onActivityResult(requestCode,resultCode,data)
   {
       if (requestCode == rq)
      {
          if (data ==  null ) return ;
         var uri = data.getData();
         playAudio(uri);
      }
      
   }

    function onDestroy()
   {
       if (med)med.release();
      med =  null ;
   }
